    11→  Clock,

    20→import {

    27→} from "@/components/ui/dialog";
    28→import {

    30→  AlertDialogAction,

    38→} from "@/components/ui/alert-dialog";

    42→import { Button } from "@/components/ui/button";
    43→
    44→interface ModelMapping {
    45→  virtual: string;
    46→  actual: string;
    47→}
    48→
    49→export function ProvidersTab() {
    50→  const { status, doToggleProvider, doAddProvider, doDeleteProvider } = useGatewayStore();

    68→    const action = isEnabled ? "disable" : "enable";
    69→    await doToggleProvider(providerId, action);
    70→    setTogglingId(null);
    71→  };
    72→
    73→  const handleDelete = async (providerId: string) => {

    75→    await doDeleteProvider(providerId);

    77→  };

   125→    updated[index] = { ...updated[index], [field]: value };

   127→  };

   129→  const isFormValid = formId.trim() && formBaseUrl.trim() && formApiKey.trim();

   137→          <p className="text-sm text-zinc-500 mt-0.5">Configure and monitor your LLM backend providers</p>

   139→        <div className="flex items-center gap-3">
   140→          <span className="text-xs text-zinc-500">
   141→            {providers.filter((p) => p.enabled && p.status === "healthy").length} of {providers.length} healthy

   145→            className="flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 hover:bg-emerald-500/20 transition-all duration-200"
   146→          >

   157→          const isDeleting = deletingId === provider.provider_id;
   158→
   159→          return (
   160→            <motion.div
   161→              key={provider.provider_id}
   162→              initial={{ opacity: 0, x: -20 }}
   163→              animate={{ opacity: 1, x: 0 }}
   164→              transition={{ delay: i * 0.08, duration: 0.4, ease: "easeOut" }}
   165→              className={`rounded-xl bg-[#11111a] border p-6 hover:border-white/[0.12] transition-all duration-300 ${
   166→                !provider.enabled
   167→                  ? "border-zinc-700/30 opacity-60"
   168→                  : provider.status === "healthy"
   169→                  ? "border-white/[0.06]"

   176→                  <div className="flex items-center gap-3">
   177→                    <div
   178→                      className={`w-10 h-10 rounded-lg flex items-center justify-center ${
   179→                        !provider.enabled
   180→                          ? "bg-zinc-500/10 border border-zinc-500/20"
   181→                          : provider.status === "healthy"
   182→                          ? "bg-emerald-500/10 border border-emerald-500/20"
   183→                          : "bg-amber-500/10 border border-amber-500/20"
   184→                      }`}
   185→                    >
   186→                      <Server
   187→                        className={`w-5 h-5 ${
   188→                          !provider.enabled
   189→                            ? "text-zinc-500"
   190→                            : provider.status === "healthy"
   191→                            ? "text-emerald-400"
   192→                            : "text-amber-400"
   193→                        }`}
   194→                      />
   195→                    </div>
   196→                    <div>
   197→                      <h3 className="text-base font-semibold text-white">{provider.provider_id}</h3>
   198→                      <div className="flex items-center gap-2 mt-0.5">
   199→                        <span
   200→                          className={`inline-flex items-center gap-1 text-[10px] uppercase tracking-wider font-bold px-2 py-0.5 rounded-full ${

   202→                              ? "bg-zinc-500/10 text-zinc-500"
   203→                              : provider.status === "healthy"
   204→                              ? "bg-emerald-500/10 text-emerald-400"
   205→                              : "bg-amber-500/10 text-amber-400"

   258→                    </div>

   262→                        {provider.total_requests > 0

   274→                        className={`text-lg font-bold ${
   275→                          provider.consecutive_failures > 0 ? "text-amber-400" : "text-zinc-400"
   276→                        }`}
   277→                      >
   278→                        {provider.consecutive_failures}
   279→                      </p>
   280→                    </div>
   281→                  </div>
   282→
   283→                  {/* Error info */}
   284→                  {provider.last_error && (
   285→                    <div className="flex items-start gap-2 p-3 rounded-lg bg-rose-500/5 border border-rose-500/10">
   286→                      <AlertTriangle className="w-4 h-4 text-rose-400 mt-0.5 shrink-0" />
   287→                      <div>
   288→                        <p className="text-xs text-rose-400 font-medium">Last Error</p>
   289→                        <p className="text-[11px] text-rose-400/70 mt-0.5">{provider.last_error}</p>
   290→                      </div>
   291→                    </div>
   292→                  )}
   293→
   294→                  {/* Cooldown info */}
   295→                  {provider.enabled && provider.status === "cooldown" && provider.cooldown_until && (
   296→                    <div className="flex items-center gap-2 p-3 rounded-lg bg-amber-500/5 border border-amber-500/10">
   297→                      <Clock className="w-4 h-4 text-amber-400 shrink-0" />
   298→                      <p className="text-xs text-amber-400">
   299→                        In cooldown — auto-recovers in ~
   300→                        {Math.max(0, Math.round(provider.cooldown_until - Date.now() / 1000))}s
   301→                      </p>
   302→                    </div>
   303→                  )}
   304→                </div>
   305→
   306→                {/* Right: Actions */}
   307→                <div className="flex flex-row lg:flex-col items-center gap-2 shrink-0">
   308→                  <button
   309→                    onClick={() => handleToggle(provider.provider_id, provider.enabled)}
   310→                    disabled={isToggling}
   311→                    className={`flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-medium transition-all duration-200 ${
   312→                      provider.enabled
   313→                        ? "bg-rose-500/10 text-rose-400 border border-rose-500/20 hover:bg-rose-500/20"
   314→                        : "bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 hover:bg-emerald-500/20"
   315→                    } disabled:opacity-50`}
   316→                  >
   317→                    {isToggling ? (
   318→                      <div className="w-3 h-3 border-2 border-current border-t-transparent rounded-full animate-spin" />
   319→                    ) : provider.enabled ? (
   320→                      <ToggleRight className="w-4 h-4" />
   321→                    ) : (
   322→                      <ToggleLeft className="w-4 h-4" />
   323→                    )}
   324→                    {provider.enabled ? "Disable" : "Enable"}
   325→                  </button>
   326→
   327→                  <AlertDialog>
   328→                    <AlertDialogTrigger asChild>
   329→                      <button
   330→                        disabled={isDeleting}
   331→                        className="flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-medium bg-rose-500/10 text-rose-400 border border-rose-500/20 hover:bg-rose-500/20 transition-all duration-200 disabled:opacity-50"
   332→                      >
   333→                        {isDeleting ? (
   334→                          <div className="w-3 h-3 border-2 border-current border-t-transparent rounded-full animate-spin" />
   335→                        ) : (
   336→                          <Trash2 className="w-3.5 h-3.5" />
   337→                        )}
   338→                        Delete
   339→                      </button>
   340→                    </AlertDialogTrigger>
   341→                    <AlertDialogContent className="bg-[#1a1a2e] border-white/10">
   342→                      <AlertDialogHeader>
   343→                        <AlertDialogTitle className="text-white">Delete Provider</AlertDialogTitle>
   344→                        <AlertDialogDescription className="text-zinc-400">
   345→                          Are you sure you want to delete <span className="text-white font-medium">{provider.provider_id}</span>?
   346→                          This action cannot be undone and will immediately remove the provider from the gateway.
   347→                        </AlertDialogDescription>
   348→                      </AlertDialogHeader>
   349→                      <AlertDialogFooter>
   350→                        <AlertDialogCancel className="bg-white/5 border-white/10 text-zinc-300 hover:bg-white/10">
   351→                          Cancel
   352→                        </AlertDialogCancel>
   353→                        <AlertDialogAction
   354→                          onClick={() => handleDelete(provider.provider_id)}
   355→                          className="bg-rose-500/20 text-rose-400 border border-rose-500/20 hover:bg-rose-500/30"
   356→                        >
   357→                          Delete
   358→                        </AlertDialogAction>
   359→                      </AlertDialogFooter>
   360→                    </AlertDialogContent>
   361→                  </AlertDialog>
   362→                </div>
   363→              </div>
   364→            </motion.div>
   365→          );
   366→        })}
   367→      </div>
   368→
   369→      {/* No Providers */}
   370→      {providers.length === 0 && (
   371→        <div className="flex flex-col items-center justify-center py-16 text-zinc-600">
   372→          <Server className="w-12 h-12 mb-3" />

   374→          <p className="text-xs mt-1">Add a provider to get started</p>
   375→          <button
   376→            onClick={() => setAddDialogOpen(true)}
   377→            className="mt-4 flex items-center gap-2 px-4 py-2 rounded-lg text-xs font-medium bg-emerald-500/10 text-emerald-400 border border-emerald-500/20 hover:bg-emerald-500/20 transition-all duration-200"
   378→          >
   379→            <Plus className="w-4 h-4" />
   380→            Add Provider
   381→          </button>
   382→        </div>

   384→
   385→      {/* Provider Config Summary */}
